Em qual linha você está?

Ao selecionar, pediremos acesso ao seu GPS para atualizar o mapa para todos os alunos.

// ========================================== // MOTOR DE FUSÃO E FILTRO ANTI-ZUMBI // ========================================== // Objeto para guardar os marcadores reais desenhados no mapa const onibusNoMapa = {}; /** * Função principal que recebe os dados "crus" (vários alunos na mesma linha), * limpa os dados velhos, calcula o centroide e move o ônibus. */ function processarDadosDaLinha(codigoLinha, corClasse, usuariosAtivos) { const tempoAtual = Date.now(); let somaLat = 0; let somaLng = 0; let quantidadeValida = 0; // 1. FILTRO ANTI-ZUMBI // Percorre todos os alunos transmitindo nesta linha for (const idUsuario in usuariosAtivos) { const dados = usuariosAtivos[idUsuario]; // Verifica a idade da informação (Ex: ignora se for mais velha que 30 segundos) const idadeDaInformacao = tempoAtual - dados.timestamp; if (idadeDaInformacao < 30000) { // 30.000 milissegundos = 30 segundos somaLat += dados.lat; somaLng += dados.lng; quantidadeValida++; } else { console.log(`Dado ignorado do usuário ${idUsuario} (Muito antigo)`); } } // 2. SE NINGUÉM FOR VÁLIDO, REMOVE O ÔNIBUS DO MAPA if (quantidadeValida === 0) { if (onibusNoMapa[codigoLinha]) { map.removeLayer(onibusNoMapa[codigoLinha]); delete onibusNoMapa[codigoLinha]; } return; // Sai da função } // 3. CÁLCULO DO CENTROIDE (FUSÃO) // Calcula a média exata de todas as latitudes e longitudes válidas const mediaLat = somaLat / quantidadeValida; const mediaLng = somaLng / quantidadeValida; // 4. ATUALIZA OU CRIA O MARCADOR NO MAPA if (onibusNoMapa[codigoLinha]) { // Se o ônibus já existe, apenas move ele suavemente para a nova posição onibusNoMapa[codigoLinha].setLatLng([mediaLat, mediaLng]); onibusNoMapa[codigoLinha].getPopup().setContent(`Linha ${codigoLinha}
Colaboradores ativos: ${quantidadeValida}`); } else { // Se não existe, cria o ícone e adiciona ao mapa const novoIcone = createBusIcon(corClasse, codigoLinha); const novoMarcador = L.marker([mediaLat, mediaLng], { icon: novoIcone }).addTo(map); novoMarcador.bindPopup(`Linha ${codigoLinha}
Colaboradores ativos: ${quantidadeValida}`); onibusNoMapa[codigoLinha] = novoMarcador; } } // ========================================== // SIMULADOR DE TESTE (Para vermos a mágica funcionando) // ========================================== // Vamos simular 3 alunos na linha "L1" (Vermelha), um deles com a internet caída (dado velho). setInterval(() => { // Simulando dados vindo do banco de dados (Firebase falso) const dadosSimuladosDoFirebase = { "aluno_01": { lat: -23.5600 + (Math.random()*0.001), lng: -46.7250, timestamp: Date.now() }, // Ativo "aluno_02": { lat: -23.5605 + (Math.random()*0.001), lng: -46.7245, timestamp: Date.now() }, // Ativo e pertinho do aluno 01 "aluno_03": { lat: -23.5000, lng: -46.7000, timestamp: Date.now() - 60000 } // ZUMBI (Há 1 minuto atrás, não vai contar) }; // Chamamos o motor de fusão para a Linha 1 (Vermelha) processarDadosDaLinha('L1', 'bg-red', dadosSimuladosDoFirebase); }, 2000); // Roda a cada 2 segundos simulando o tempo real
